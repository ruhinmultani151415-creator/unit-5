//demonstrate the difference between structure and union
#include <stdio.h>

struct student
{
int a;
int b;
};

union data
{
int a;
int b;
};

int main()
{
struct student s;
union data u;

s.a = 10;
s.b = 20;

u.a = 10;
u.b = 20;

printf("Structure: a = %d, b = %d\n", s.a, s.b);
printf("Union: a = %d, b = %d\n", u.a, u.b);

return 0;

}
