//copy one file another file
#include <stdio.h>

int main()
{
FILE *fp1, *fp2;
int ch;

fp1 = fopen("data.txt", "r");

if (fp1 == NULL)
{
    printf("Source file not found");
    return 1;
}

fp2 = fopen("copy.txt", "w");

if (fp2 == NULL)
{
    fclose(fp1);
    return 1;
}

while ((ch = fgetc(fp1)) != EOF)
{
    fputc(ch, fp2);
}

fclose(fp1);
fclose(fp2);

printf("File copied successfully");

return 0;

}
