//write a string in greencity.txt and Print It in Reverse Order in cleancity.txt
#include <stdio.h>
#include <string.h>

int main()
{
FILE *fp1, *fp2;
char str[100];
int i;

fp1 = fopen("greencity.txt", "w");

if (fp1 == NULL)
    return 1;

printf("Enter a string: ");
fgets(str, sizeof(str), stdin);

str[strcspn(str, "\n")] = '\0';

fprintf(fp1, "%s", str);
fclose(fp1);

fp1 = fopen("greencity.txt", "r");
fp2 = fopen("cleancity.txt", "w");

if (fp1 == NULL || fp2 == NULL)
{
    if (fp1 != NULL)
        fclose(fp1);

    if (fp2 != NULL)
        fclose(fp2);

    return 1;
}

if (fgets(str, sizeof(str), fp1) != NULL)
{
    for (i = (int)strlen(str) - 1; i >= 0; i--)
    {
        fputc(str[i], fp2);
    }
}

fclose(fp1);
fclose(fp2);

printf("Reversed string written to cleancity.txt");

return 0;

}
