//Store Numbers 1 to 10 in DATA.txt, Then Separate Odd and Even Numbers
#include <stdio.h>

int main()
{
FILE *fp, *odd, *even;
int i;

fp = fopen("DATA.txt", "w");

if (fp == NULL)
    return 1;

for (i = 1; i <= 10; i++)
{
    fprintf(fp, "%d ", i);
}

fclose(fp);

fp = fopen("DATA.txt", "r");
odd = fopen("ODD.txt", "w");
even = fopen("EVEN.txt", "w");

if (fp == NULL || odd == NULL || even == NULL)
{
    if (fp != NULL)
        fclose(fp);

    if (odd != NULL)
        fclose(odd);

    if (even != NULL)
        fclose(even);

    return 1;
}

while (fscanf(fp, "%d", &i) == 1)
{
    if (i % 2 == 0)
    {
        fprintf(even, "%d ", i);
    }
    else
    {
        fprintf(odd, "%d ", i);
    }
}

fclose(fp);
fclose(odd);
fclose(even);

printf("Odd and even numbers stored successfully");

return 0;

}
