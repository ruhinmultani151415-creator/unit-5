//write content in a file and read it
#include <stdio.h>

int main()
{
FILE *fp;
char ch;

fp = fopen("data.txt", "w");

if (fp == NULL)
    return 1;

fprintf(fp, "Hello World");
fclose(fp);

fp = fopen("data.txt", "r");

if (fp == NULL)
    return 1;

while ((ch = fgetc(fp)) != EOF)
{
    printf("%c", ch);
}

fclose(fp);

return 0;

}
